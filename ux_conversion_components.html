// Dynamic Pre-filled WhatsApp Routing Script
(function() {
  var waPhone = '6281316333001';
  var stickyLink = document.getElementById('sticky-cta-btn');
  
  // Deteksi posisi scroll pengguna untuk menyesuaikan konteks pesan
  function updateContextualWhatsApp() {
    if (!stickyLink) return;
    var scrollY = window.scrollY;
    var paketSection = document.getElementById('paket');
    var jadwalSection = document.getElementById('jadwal');
    var pesan = "Bismillah, Assalamu'alaikum Admin H2C. Saya ingin konsultasi ketersediaan seat & brosur resmi Umroh.";

    if (jadwalSection && scrollY >= jadwalSection.offsetTop - 300) {
      pesan = "Bismillah, saya sedang melihat tabel Jadwal Keberangkatan Umroh H2C. Mohon info seat yang masih terbuka untuk kloter terdekat.";
    } else if (paketSection && scrollY >= paketSection.offsetTop - 300) {
      pesan = "Bismillah, saya sedang memilih Paket Umroh di website H2C. Mau tanya rincian promo kunci seat DP Rp 5 Juta & fasilitas hotelnya.";
    }

    stickyLink.href = "https://wa.me/" + waPhone + "?text=" + encodeURIComponent(pesan);
  }

  window.addEventListener('scroll', updateContextualWhatsApp, { passive: true });
})();
